import type { ExtensionAPI, ExtensionContext } from "@earendil-works/pi-coding-agent";
import { existsSync } from "node:fs";
import { createConnection } from "node:net";
import { execFileSync } from "node:child_process";

const SOCKET_PATH = "/tmp/agent-visor.sock";
const SOCKET_TIMEOUT_MS = 100;
const HEARTBEAT_INTERVAL_MS = 10_000;

function controllingTTY(): string | undefined {
  try {
    const value = execFileSync("/usr/bin/tty", [], {
      encoding: "utf8",
      stdio: ["inherit", "pipe", "ignore"],
      timeout: 100,
    }).trim();
    if (!value || value === "not a tty") return undefined;
    return value.replace(/^\/dev\//, "");
  } catch {
    return undefined;
  }
}

const tty = controllingTTY();

// The runtime's own "nothing is running" flag: false while an agent run,
// automatic retry, auto-compaction retry, or queued continuation is in
// flight. Reported as metadata so Agent Visor can repair a phase whose
// lifecycle event was lost in transit — one dropped `Stop` otherwise pins a
// finished session to Working until the user's next prompt. Probed
// defensively: an older Pi runtime may not expose `isIdle`, and a stale
// extension context throws instead of answering. Both cases report no flag,
// which preserves the previous phase-neutral heartbeat.
function runtimeIsIdle(ctx: ExtensionContext): boolean | undefined {
  const probe = (ctx as { isIdle?: () => boolean }).isIdle;
  if (typeof probe !== "function") return undefined;
  try {
    return probe.call(ctx) === true;
  } catch {
    return undefined;
  }
}

interface ReportOptions {
  tool?: string;
  toolUseId?: string;
  idle?: boolean;
}

function report(
  ctx: ExtensionContext,
  lifecycleEvent: string,
  status: string,
  options: ReportOptions = {},
): void {
  if (ctx.mode !== "tui" || !existsSync(SOCKET_PATH)) return;

  const payload = {
    session_id: ctx.sessionManager.getSessionId(),
    session_file: ctx.sessionManager.getSessionFile(),
    cwd: ctx.cwd,
    event: lifecycleEvent,
    status,
    pid: process.pid,
    tty,
    tool: options.tool,
    tool_use_id: options.toolUseId,
    is_idle: options.idle,
    agent: "pi",
  };

  const socket = createConnection(SOCKET_PATH);
  socket.setTimeout(SOCKET_TIMEOUT_MS);
  socket.on("connect", () => socket.end(JSON.stringify(payload)));
  socket.on("timeout", () => socket.destroy());
  socket.on("error", () => socket.destroy());
  socket.unref();
}

export default function (pi: ExtensionAPI) {
  let heartbeatTimer: ReturnType<typeof setInterval> | undefined;

  function stopHeartbeat(): void {
    if (heartbeatTimer === undefined) return;
    clearInterval(heartbeatTimer);
    heartbeatTimer = undefined;
  }

  function startHeartbeat(ctx: ExtensionContext): void {
    stopHeartbeat();
    if (ctx.mode !== "tui") return;
    heartbeatTimer = setInterval(() => {
      report(ctx, "SessionHeartbeat", "alive", { idle: runtimeIsIdle(ctx) });
    }, HEARTBEAT_INTERVAL_MS);
    heartbeatTimer.unref();
  }

  pi.on("session_start", (_event, ctx) => {
    report(ctx, "SessionStart", "idle");
    startHeartbeat(ctx);
  });

  pi.on("agent_start", (_event, ctx) => {
    report(ctx, "AgentStart", "processing");
  });

  pi.on("tool_execution_start", (event, ctx) => {
    report(ctx, "PreToolUse", "running_tool", {
      tool: event.toolName,
      toolUseId: event.toolCallId,
    });
  });

  pi.on("tool_execution_end", (event, ctx) => {
    report(ctx, "PostToolUse", "processing", {
      tool: event.toolName,
      toolUseId: event.toolCallId,
    });
  });

  pi.on("session_before_compact", (_event, ctx) => {
    report(ctx, "PreCompact", "compacting");
  });

  // The closing boundary of a compaction. A manual `/compact` runs outside
  // an agent run, so it never reaches `agent_settled`: without this event the
  // row would stay Compacting until the user's next prompt. Auto compaction
  // happens inside a run, where the runtime still reports busy and the turn's
  // own lifecycle events stay authoritative.
  pi.on("session_compact", (_event, ctx) => {
    const idle = runtimeIsIdle(ctx);
    report(ctx, "PostCompact", idle === false ? "processing" : "idle", { idle });
  });

  // agent_end is intentionally not a completion signal: Pi may still retry,
  // compact, or run queued follow-ups. agent_settled is the user-visible
  // Ready boundary.
  pi.on("agent_settled", (_event, ctx) => {
    report(ctx, "Stop", "waiting_for_input");
  });

  pi.on("session_shutdown", (event, ctx) => {
    stopHeartbeat();
    if (event.reason !== "reload") {
      report(ctx, "SessionEnd", "ended");
    }
  });
}
